import unittest
import requests

class UserLifecycleSystemTest(unittest.TestCase):
    
    @classmethod
    def setUpClass(cls):
        """Runs once before the entire system test suite starts."""
        cls.base_url = "http://localhost:5000/api"
        cls.credentials = {
            "username": "system_tester",
            "password": "SecurePassword123!"
        }

    def test_01_full_user_lifecycle_flow(self):
        """Validates the entire end-to-end user workflow sequentially."""
        
        # Step 1: Execute User Registration
        register_url = f"{self.base_url}/register"
        reg_response = requests.post(register_url, json=self.credentials)
        
        # Verify Registration System Results
        self.assertEqual(reg_response.status_code, 201)
        self.assertIn("User created successfully", reg_response.json().get("message"))

        # Step 2: Execute User Login (Dependent on Step 1)
        login_url = f"{self.base_url}/login"
        login_response = requests.post(login_url, json=self.credentials)
        
        # Verify Login System Results
        self.assertEqual(login_response.status_code, 200)
        self.assertIn("token", login_response.json())

    def test_02_login_rejection_with_invalid_password(self):
        """Negative Flow: Validates security system blocks unauthorized access."""
        login_url = f"{self.base_url}/login"
        bad_credentials = {
            "username": self.credentials["username"],
            "password": "WrongPassword"
        }
        
        response = requests.post(login_url, json=bad_credentials)
        
        # Verify Security System Rejection
        self.assertEqual(response.status_code, 401)
        self.assertIn("Invalid credentials", response.json().get("error"))

if __name__ == "__main__":
    # Execute the system tests
    unittest.main()
